# Kibana Dev Tools > Console. Paste the whole file and run each request with the green arrow, top to bottom.
# Lines starting with # are comments. Every request below was run against an Elasticsearch Serverless project.
# Labels: Open Food Facts product photos (data ODbL, images CC BY-SA). The text below is real output of the
# document_extraction call, so no editing is needed.

# ---------------------------------------------------------------------------------------------------------
# 1. Is the reader there? Look for task_type document_extraction, inference_id .jina-reader. Nothing to create.
GET _inference/_all

# ---------------------------------------------------------------------------------------------------------
# 2. Extract: one call, image in, Markdown out. The filter keeps the response short (content only).
POST _inference/document_extraction/.jina-reader?filter_path=document_extraction.content
{
  "input": [
    {
      "content": {
        "type": "image",
        "format": "url",
        "value": "https://images.openfoodfacts.org/images/products/807/680/952/3547/nutrition_de.380.full.jpg"
      }
    }
  ],
  "task_settings": { "parameters": { "preset": "ocr" } }
}

# ---------------------------------------------------------------------------------------------------------
# 3. The call stores nothing, so create an index. semantic_text needs no inference endpoint: on Serverless the
#    default is .jina-embeddings-v5-text-small. copy_to feeds the label text into it.
PUT food_labels_demo
{
  "mappings": {
    "properties": {
      "product_name": { "type": "text" },
      "language":     { "type": "keyword" },
      "source_url":   { "type": "keyword" },
      "content":      { "type": "text", "copy_to": "meaning" },
      "meaning":      { "type": "semantic_text" }
    }
  }
}

# 4. Index three labels (German, French, English): the Markdown the reader returned, as one-line JSON strings.

# meaning should show "inference_id": ".jina-embeddings-v5-text-small"
GET food_labels_demo/_mapping

# ---------------------------------------------------------------------------------------------------------
# 5. Search by words. Only labels that contain the word match. No label prints "sugar": the German one says Zucker.
POST food_labels_demo/_search
{
  "query": { "match": { "content": "sugar" } },
  "_source": ["product_name", "language"]
}

# Now the word as the German label prints it: one hit.
POST food_labels_demo/_search
{
  "query": { "match": { "content": "Zucker" } },
  "_source": ["product_name", "language"]
}

# ---------------------------------------------------------------------------------------------------------
# 6. Search by meaning. Same search API, only the query changes. An English question ranks the French label first.
#    It ranks every label by closeness (it does not filter), so read the scores.
POST food_labels_demo/_search
{
  "query": { "semantic": { "field": "meaning", "query": "fresh dairy cheese" } },
  "_source": ["product_name", "language"]
}

POST food_labels_demo/_search
{
  "query": { "semantic": { "field": "meaning", "query": "dark chocolate high in fat" } },
  "_source": ["product_name", "language"]
}

# ---------------------------------------------------------------------------------------------------------
# 7. The same data with ES|QL (in Discover, pick ES|QL mode and paste only the query text).
POST _query?format=txt
{
  "query": "FROM food_labels_demo | STATS labels = COUNT(*) BY language | SORT language"
}

POST _query?format=txt
{
  "query": "FROM food_labels_demo METADATA _score | WHERE MATCH(meaning, \"fresh dairy cheese\") | SORT _score DESC | KEEP product_name, language, _score | LIMIT 3"
}

# ---------------------------------------------------------------------------------------------------------
# 8. Clean up when done.
DELETE food_labels_demo
